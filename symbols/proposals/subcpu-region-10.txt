# region 10: 0x03111A..0x0331D4
# Tone-database (ToneDB) record resolution and per-part cache binding, plus the
# 0x8x-opcode inter-CPU "read tone data" service that answers main->sub queries
# out of the reply staging buffer at 0x045210/0x045216.

[DATA] 0x00FAEE-0x00FB1D  ToneDB_Query_Dispatch jump table (24 x u16)
; 24 little-endian 16-bit displacements added to 0x0317CF by the switch at
; 0x0317B9..0x0317CE.  Index = (request[+2] & 0x7F), bounds-checked 0..0x17.
; Resolved targets: 0->0x0317CF 1->0x0317EA 2,3->0x031A51(nop) 4->0x03182A
; 5->0x03184F 6->0x03186D 7->0x03188B 8->0x0318AA 9->0x0318C9 10->0x0318E7
; 11->0x031905 12->0x031915 13->0x031949 14->0x03196E 15->0x031993
; 16->0x0319B2 17->0x0319D0 18->0x0319EE 19->0x031A19 20,21,22->0x031A51(nop)
; 23->0x031A44.

[DATA] 0x00FB1E-0x00FB2D  VoiceParam_Query_Dispatch jump table (8 x u16)
; Displacements added to 0x031B5B.  Index = request[+0] & 7, taken when bit 3 of
; the status byte is CLEAR (the "query, reply with data" family).
; 0->0x031B5B 1->0x031B63 2->0x031BC8 3->0x031C26 4->0x031C84 5->0x031D16
; 6->0x031D4D 7->0x031DA1.

[DATA] 0x00FB2E-0x00FB3D  VoiceParam_Action_Dispatch jump table (8 x u16)
; Displacements added to 0x031AA1.  Index = request[+0] & 7, taken when bit 3 of
; the status byte is SET (the "action, no reply" family).
; 0->0x031AA1 1->0x031AA9 2->0x031AB9 3->0x031AE5 4->0x031B11 5->0x031B21
; 6->0x031B29 7->0x031B31.

[DATA] 0x00FB4E-0x00FB55  Partial-present bit-mask table (4 x u16, mirrored)
; { 0x0001, 0x0004, 0x0010, 0x0040 } indexed by partial index * 2, read by
; Partial_Find_Rank (0x032082) via XIY.  These are bits 0/2/4/6 of the tone
; record's partial-present byte at tonerec+0x11 -- the same four bits tested
; individually by Partial_Build_Present_Word (0x032B1E).  The table is stored
; twice (8 entries) but only indices 0..3 are ever used.
