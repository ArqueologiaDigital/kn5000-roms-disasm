# region 8: 0x02C8E4..0x02EC37
# The tone-generator (IC303) register-marshalling layer: velocity/key-on dispatch, the voice
# "query" packet builders, every ToneGen_Write* primitive, and the VoiceParam_* staging-buffer
# packers. Everything here talks to exactly one piece of hardware -- the TG at 0x100000
# (address latch) / 0x100002 (data). Nothing in this region touches the inter-CPU latches
# (0x120000), the keybed (0x110000) or the DSPs (0x130000); verified by scanning every
# absolute operand in the range.
#
# THE TG ACCESS IDIOM (appears ~110 times; every "*_NopCont*" placeholder is part of it):
#     res 7,(0x18)                 ; P6 bit 7 low  = address-latch phase
#     add WA,<regbase>             ; WA = regbase + voice channel
#     ld (0x100000),WA             ; latch TG register address
#     nop
#     set 7,(0x18)                 ; P6 bit 7 high = data phase
#     ld (0x100002),<value>        ; TG register data
#     jr $+2 / nop / nop / nop     ; ~4-cycle settle delay before the next access
# So SFR P6 bit 7 is the TG address/data phase strobe. The "jr $+2" targets are what the
# converter turned into __jrt_nop_xxxxxx / *_NopCont* labels; they are pure padding and are
# NOT worth individual names.
#
# TG REGISTER BANKS (each bank is 0x40 registers, one per voice channel; channel 0x00-0x3F):
#   0x0000+ch key-on / note word (0x7E00 = free the voice)   0x0040+ch, 0x0080+ch (bit15 =
#   load strobe), 0x00C0+ch, 0x0100+ch, 0x0140+ch, 0x0180+ch, 0x01C0+ch, 0x0200..0x0205 and
#   0x0C00..0x0C05 and 0x0E00 (global, not per-voice), 0x0400/0x0440/0x0480/0x04C0/0x0500+ch,
#   0x0540/0x0580/0x05C0+ch (bit15 = load strobe), 0x0600/0x0640+ch,
#   0x0800+ch ... 0x0A40+ch (envelope/level bank).
#
# THE VOICE-PARAMETER SHADOW BLOCK (the XBC/XWA argument of every ToneGen_Write*), 0x44 bytes.
# Offset -> TG register base, established by reading every write site in the region:
#   +0x02->0x0040  +0x04->0x0080  +0x06->0x00C0  +0x08->0x0100  +0x0A->0x0140  +0x0C->0x0180
#   +0x0E->0x0400  +0x10->0x0440  +0x12->0x0480  +0x14->0x04C0  +0x16->0x0500  +0x18->0x0800
#   +0x1A->0x0840  +0x1C->0x0880  +0x1E->0x08C0  +0x20->0x0900  +0x22->0x0940  +0x24->0x0980
#   +0x26->0x09C0  +0x28->0x0A00  +0x2A->0x0A40  +0x2C->0x0800  +0x2E->0x0840  +0x30->0x0900
#   +0x32->0x0940  +0x34->0x09C0  +0x36->0x0A00  +0x38->0x01C0  +0x3A->0x0540  +0x3C->0x0580
#   +0x3E->0x05C0  +0x40->0x0600  +0x42->0x0640
# The live instance of this block is DRAM 0x0451CC (Voice_Release/Voice_Cut pass it literally);
# the boot/default instance is DRAM 0x2AA4, filled from ROM 0x00F8D5 by DSP_CONFIG_INIT.
#
# DRAM STRUCTURES USED THROUGHOUT (all verified by the index arithmetic in this region):
#   0x041368 + part*0x11F   part descriptor, 0x11F bytes, part = 0..0x19 (26 parts)
#       +0x00 w  state bits: bit0/bit1 = layer select, bit2 = loaded, (&3)!=0 means "in use"
#       +0x06 l  pointer to the 0x66-byte TONE HEADER; header+0x10 & 0xC0 = PART MODE
#                (0x00 / 0x40 / 0x80 / 0xC0) which selects the voice-layer topology
#       +0x0A w  routing word: bit15 tested by VOICEPARAM_BUILD_TONE_STAGING, bit14 set from
#                tone-header +0x5D bit7
#       +0x19/+0x1A b  two bytes passed to VoiceInit_Dispatcher (0x032938)
#       +0x6E + i*0x25 (i=0..3)  sub-slot descriptor; +0x00 l = pointer to a 0x51-byte element
#                record, +0x36 b &7 = element group, +0x4D+j (j=0..3) = per-slot params
#   0x04308E + ch*0x47      per-voice-channel struct, 0x47 bytes, ch = 0..0x3F
#       +0x01 w flags (bit8 = "hold" path), +0x03 b stage, +0x05 b bit7 = active,
#       +0x23 l back-pointer to the owning part record, +0x2D w = the key-on word written to
#       TG register (0x0000+ch), +0x2F w = 0 gate for the key-on write, +0x46 b
#   0x044FCE                TONE STAGING BUFFER built by VOICEPARAM_BUILD_TONE_STAGING:
#       +0x0000 (0x66)          tone header, copied from part+0x06
#       +0x0066 + i*0x51 (i<4)  element records, copied from part+0x6E+i*0x25
#       +0x01AA + i*0x0B (i<4)  sub-slot records
#       +0x01D8 + k (k<8)       tone-option bytes, defaults copied from ROM 0x00F95D
#       +0x01E0 + i*5 + j       per-slot params, copied from part sub-slot +0x4D
#       +0x01F4 + k*5           custom-tone unit params
#       +0x01FE (= 0x0451CC)    the TG shadow block described above
#   0x045310 / 0x045314 / 0x04531C   three long pointers to the loaded sound-data image; the
#       0x045314 block is a table of offsets (+0x50, +0x64, +0x80, +0x8C, +0x94 select wave
#       catalogues; +0xEA/+0xEE/+0xF0 are record LENGTHS), 0x04531C is the custom-tone bank.

[ROUTINE] 0x02C8E4 VOICE_SETVELOCITY  confidence=high
; Already named in symbols/subcpu_symbols_reference.txt and in the LLVM build; entry kept
; unchanged, this entry only adds documentation.
; The note-on workhorse. Inputs: A = part index (0..0x19), C = MIDI note, E = velocity,
; plus one word pushed by the caller (Voice_NoteOn pushes MIDI status & 7 = the "source"
; tag). Stack frame: 20 bytes of scratch + saved 0xFA/0xFB byte temporaries; the caller's
; pushed word lands at (XSP+26); returns with retd 2.
; It reads the part descriptor pointer at 0x04136E + part*0x11F, dereferences it and
; switches on tone-header byte +0x10 masked with 0xC0 -- the PART MODE:
;   0xC0 -> falls straight through to VOICE_SETVELOCITY_EXIT: NOTHING IS ALLOCATED AND NO TG
;           REGISTER IS TOUCHED. This is a silent no-op path.
;   0x80 -> VOICE_SETVELOCITY_MODE80 (2 voice slots, Voice_NoteOn_Type1 @0x02C2C0)
;   0x40 -> VOICE_SETVELOCITY_MODE40 (4 slots, Voice_NoteOn_Type2 @0x02BF1B)
;   0x00 -> in-line mode-00 path      (4 slots, Voice_NoteOn_Type4 @0x02BA2C)
;   anything else -> exit, also a no-op.
; The chosen Voice_NoteOn_TypeN fills the 20-byte scratch frame; scratch[0x0A + i] is the TG
; channel allocated for slot i (0xFF/>=0x40 means "no slot"). Two loops then follow: see
; VOICE_SETVELOCITY_MODE00_KEYON_LOOP and VOICE_SETVELOCITY_MODE00_PITCHFIX_LOOP.
; On exit it clears bits 2..3 of the part word at 0x041368 + part*0x11F.
; Callers: VOICE_NOTEON (0x02CF97) and the MIDI/keybed note path.

[LABEL] 0x02C96B VOICE_SETVELOCITY_MODE00_KEYON_LOOP  replaces=LABEL_02C96B  confidence=high

; Voice_Init_Type2 (0x02C0B6) or Voice_Init_Type4 (0x02B4E3) depending on bit1 of the voice
; struct word at 0x04308F + ch*0x47.

[LABEL] 0x02C9FA VOICE_SETVELOCITY_MODE00_KEYWORD_LOOP  replaces=LABEL_02C9FA  confidence=high
