# region 11: 0x0331D5..0x0354C9
# The DSP effect-algorithm / output-routing layer plus the MIDI event dispatcher.  Everything
# here works on ONE data structure -- the 26-entry PART table at 0x041368 -- and turns the
# "algorithm type" nibble of a part's tone descriptor into (a) per-output-slot routing words,
# (b) per-channel DSP coefficient words looked up out of the constant tables at 0x010FCE /
# 0x011016 / 0x011346 / 0x011478 / 0x011E16, and (c) a linear DSP1 image at (0x04531C) that is
# pushed to the main CPU with InterCPU_E1_DMA_Transfer.  The tail of the region is the MIDI
# ring-buffer reader and the status-byte dispatcher (MIDI_Dispatch, 0x034D93) that feeds the
# note-on / CC / program-change handlers in region 6.
#
# INTEGRATION NOTE (read this first, it saves a wasted pass):
#   The AUTHORITATIVE LLVM build already carries a real name for EVERY address in this range
#   -- 209 symbols, ZERO LABEL_ placeholders (verified with
#   `llvm-nm -n rebuilt_ROMs/kn5000_subprogram_v142.llvm.elf`).
#   symbols/subcpu_symbols_reference.txt is the file that lags: 174 of its 209 entries in this
#   range are still LABEL_xxxxxx (the 35 already-real ones are all in 0x034A4A..0x034FE4).
#   Those 174 need no thought -- copy the ELF name verbatim; the `replaces=` field below gives
#   the exact placeholder for each.  What this file CONTRIBUTES is (a) doc headers for the
#   existing names, (b) 19 genuinely NEW names for computed-jump arms and one leaf subroutine
#   that live inside blocks the converter emitted as `.byte` data and therefore have no symbol
#   at all, and (c) the decoded jump-offset tables that make those arms readable.
#   NOTHING in this file renames an existing non-LABEL_ symbol.
#
# ★ THE "*_TableData" / "*_JumpTable*" / "*_Data" BLOCKS IN THIS REGION ARE CODE, NOT DATA.
#   The converter emitted 14 `.byte` runs here.  Every one of them disassembles cleanly as
#   TLCS-900H from its first byte and is entered by a computed `jp T,XIX+DE` whose displacement
#   comes from a word table in the program image (see [DATA] entries for 0x00FB66..0x00FBE3).
#   Affected labels: DSP_AlgoType_Dispatch1_TableData, DSP_AlgoType_Dispatch2_TableData,
#   DSP_AlgoType_Dispatch3_TableData, Algo_SubTable_JumpTable1/2/3,
#   Voice_ProgChange_TableData, Audio_Process_Init_Data, RingBuf_ReadByte_Data,
#   Audio_CmdHandler_ConstData, DSP_FlushAllSlots_Data, DSP_WriteAlgoBuffer_Data.
#   The "TableData"/"Data"/"ConstData" suffixes are actively misleading; see [UNCERTAIN].
#
# STRUCTURES USED THROUGHOUT (all read off the code in this region)
#
#   PART table  0x041368, 26 entries x 0x11F bytes.  Index = part 0..0x19.  Absolute addresses
#   that appear as `lda XBC,0x0413xx` are just 0x041368 + field offset:
#     +0x00 w  0x041368  status word.  b0 = "this part is the panel/edit target" gate
#     +0x02 w  0x04136A  flags word.  b9 (0x0200) = algorithm type 0x0A marker
#     +0x04 w  0x04136C  cleared by Voice_ActiveFlag_Set
#     +0x06 L  0x04136E  LONG POINTER to the tone/effect DESCRIPTOR (see below).  This single
#                        field is dereferenced by ~30 routines in this region.
#     +0x0A w  0x041372  flags word 2.  b15 = "descriptor present / effect engaged"
#                        (gates almost every algorithm path), b14 = prog-change param-4 flag
#     +0x19 b  0x041381  program-change value 1   +0x1A b 0x041382 program-change value 2
#     +0x1B b  0x041383  and +0x1C b 0x041384  = 0x40 after Voice_NoteState_Clear
#     +0x1D w  0x041385  note-state word, zeroed by Voice_NoteState_Clear
#     +0x1F b  0x041387  +0x20 b 0x041388  note-state bytes, zeroed likewise
#     +0x27 + 4*i + 0x10*j   4x3 array of 4-byte note-state records (i<4, j<3);
#                        bytes +0x00/+0x02/+0x03 of each are cleared by Voice_NoteState_Clear
#                        and VoiceSlot_ClearAll
#     +0x57 + 4*ch w  DSP_AlgoType_Dispatch1 result for channel ch (0/1)
#     +0x59 + 4*ch w  DSP_AlgoType_Dispatch2 result for channel ch (0/1)
#     +0x5F w  0x0413C7  DSP_ChanFreq_WritePacket1 result
#     +0x61 w  0x0413C9  DSP_ChanFreq_WritePacket2 result
#     +0x63 b  0x0413CB  +0x64 b 0x0413CC  +0x65 b 0x0413CD  three algorithm coefficient bytes
#     +0x67 b  0x0413CF  Voice_ActiveFlag_CheckAndLoad result
#     +0x68..+0x6C b  0x0413D0..0x0413D4  five DSP slot-parameter bytes (one per output slot)
#     +0x6E + 0x25*s  four OUTPUT-SLOT records, s = 0..3, 0x25 bytes each:
#          +0x04 L  long pointer to the slot's DSP parameter block (DSP_WriteAlgoBuffer)
#          +0x06 b  slot flags; b0..b2 kept, b5 set by Voice_ProgChange_TableData
#          +0x1A/+0x1C/+0x1E w  THREE ROUTING WORDS (written by RoutingFlag_Encode and the
#                   SlotRouting_* family; zeroed wholesale by VoiceSlot_ClearAll)
#          +0x20/+0x21/+0x22 b  three per-slot DSP parameter bytes (VoiceNoteParam_UpdateLoop)
#          +0x26 b  source byte for the SlotRouting_Chan1 write
#          +0x27 b  algorithm gate byte read by EFF_RoutingInit (mask 0x0A vs 0x05)
#          +0x38 b  source byte for the SlotRouting_Chan3 write
#
#   DESCRIPTOR (whatever part+0x06 points at) -- the per-part effect algorithm record:
#     +0x10 b  top two bits pick the re-init path in Voice_ProgChange:
#              0xC0 -> VoiceSlot_FullInit, 0x40 -> VoiceSlot_AltInit, 0x80/0x00 -> nothing
#     +0x5D b  LOW NIBBLE = ALGORITHM TYPE (values 0..0x0B are the ones dispatched on);
#              bit 6 = a secondary mode flag consulted by the Algo67_/Dispatch3 paths
#     +0x5E,+0x5F,+0x60,+0x61,+0x62,+0x63,+0x64,+0x65 b  algorithm parameter indices, used as
#              subscripts into the constant tables below
#
#   MIDI EVENT RING (fixed globals, 4096-byte payload):
#     0x2B0D w  write index      0x2B0F w  read index      0x2B11 w  byte count
#     0x2B13    4096-byte payload; both indices masked with 0x0FFF
#   Producer = Audio_CmdHandler_00_1F (inter-CPU command class 0x00-0x1F).
#   Consumer = MIDI_Dispatch via RingBuf_ReadByte.
#   Per-message-class staging buffers, 6 bytes each, at 0x2AE8, 0x2AEE, 0x2AF4, 0x2AF8,
#   0x2AFC, 0x2B01, 0x2B05, 0x2B09 (note-off, note-off-ext, note-on, CC, prog-change,
#   channel-pressure, pitch-bend, system).
#
#   DSP1 IMAGE:  (0x04531C) is a long pointer to a ~0x72AA-byte work image.  Notable offsets:
#     +0x0010 + 0x01D6*slot   per-slot 0xD5-word algorithm block  (DSP_WriteAlgoBuffer)
#     +0x49A7 + 2*v           128 per-voice words                 (DSP_FlushAllSlots)
#     +0x4980                 base used by DSP_InitChannelSlot / DSP_Reinit_VoiceSlots
#     +0x4AA7 + 0x50*ch       per-channel 0x1D-word slot record   (DSP_InitChannelSlot)
#     +0x72A8 w               the one's-complement checksum from DSP_WriteCount_Compute
#   The whole image is shipped to the main CPU by
#     InterCPU_E1_DMA_Transfer(src=0x007800, len=0x72AA, dst=0x001E0000)  -- 0x020DB3.

[ROUTINE] 0x0331D5 RoutingFlag_Encode  replaces=LABEL_0331D5  confidence=high
; Writes ONE 16-bit routing word into an output-slot record.
; Inputs: XWA = base address of the output-slot record (part+0x6E+0x25*slot),
;         E   = the DSP channel/slot selector used for the mask tests,
;         XBC = pointer to the slot's gate record; only (XBC+1) -- an 8-bit enable byte -- is
;               read,
;         stack arg at (XSP+0x0C) = the 8-bit payload value,
;         stack arg at (XSP+0x0E) = L = which of the three routing words to write (0/1/2).
; Output: (XWA + 0x1A + 2*L) = payload | one of six constants.  Nothing else is touched.
; Mechanism: (XBC+1) is ANDed with SlotEnableMaskA[E] (0x00FB56 = {1,4,0x10,0x40}).  If that
; is zero the routine returns without writing anything -- that is the "slot not routed" case.
; It is then ANDed with SlotEnableMaskB[E] (0x00FB5E = {2,8,0x20,0x80}); non-zero selects the
; constants {0x00C0, 0xC000, 0x3300} for L = {0,1,2}, zero selects {0x0040, 0x4000, 0x1100}.
; So the enable byte carries TWO bits per slot: bit(2s) = routed at all, bit(2s+1) = which of
; the two level/pan encodings to stamp into the word.
; Caller: EFF_RoutingInit only (0x033463).  Callee-cleans 4 bytes of arguments (retd 0x0004).

; Rebuilds the routing words of ONE output slot of ONE part for all four DSP sub-slots.
; Inputs: A = part index (0..0x19), C = output-slot index (0..3), E = routing word selector

[ROUTINE] 0x033557 AlgoType_StateWrite  replaces=LABEL_033557  confidence=high
; The master "an algorithm changed, recompute everything" entry for one part.
; Input: A = part index; C = a flag (0 = full recompute, non-zero = refresh only).
; Reads the algorithm type = descriptor[0x5D] & 0x0F and splits three ways:
;   type == 7 and C == 0 : sets part+0x0A bit 15, then writes the coefficient triple
;       part+0x63/0x64/0x65 = {0x0F,0x4F,0x96} if part+0x0A bit 14 is set, else {1,1,1}, then
;       runs the full six-call recompute (Dispatch1/2 for channels 0 and 1, then
;       DSP_ChanFreq_WritePacket1 and 2).
;   type == 7 and C != 0 : refresh only -- masks part+0x63 with 0x05 and optionally sets bit 1.
;   type != 7            : sets part+0x0A bit 15 only if bit 14 is set, runs the same six-call
;       recompute, then sets part+0x02 bit 9 when the type is 0x0A and clears it otherwise;
;       if bit 14 was clear it instead CLEARS part+0x0A bit 15, clears part+0x02 bit 9 and
;       zeroes the coefficient triple.
; So part+0x0A bit 14 ("a real algorithm is loaded") is the master enable and bit 15 is the
; derived "engaged" flag that every other routine in this region tests.
; Callers: VoiceSlot_FullInit (0x03489D), VoiceSlot_AltInit (0x034977) and one site outside
; the region.
