# region 3: 0x02219F..0x024068
# Per-tick voice servicing and the note-on/note-off lifecycle: the tone-generator
# active-voice poll that recycles finished voices, voice allocation across four
# layer slots, note-off enumeration, velocity/pan/level scaling, the pitch
# accumulator, and the per-voice filter (TVF) register builder.
#
# NOTE FOR THE INTEGRATOR: every address in this region already carries a real
# name in the authoritative LLVM build (rebuilt_ROMs/kn5000_subprogram_v142.llvm.elf).
# Nothing below renames anything. The names quoted are exactly the ELF's; the
# 'replaces=' column exists because symbols/subcpu_symbols_reference.txt is STALE
# here and still holds LABEL_xxxxxx at most of these addresses. The deliverable is
# the documentation plus that mechanical curated-file sync. Where the curated file
# already holds a different real name the disagreement is recorded under
# [UNCERTAIN] instead - several of those curated names look BETTER than the ELF's.
# Verified before writing: both toolchains rebuild at 100.00% (compare_roms.py).

[DATA] 0x00F603-0x00F632  Voice-pool priority lists (0xFF-terminated byte lists)
; Four variable-length lists of pool indices in allocation-priority order:
;   0x00F603..0x00F60A  06 05 02 04 03 01 00 FF   used by NoteChain_FindNode_B
;   0x00F60B..0x00F619  86 85 06 05 84 83 82 04 03 02 81 80 01 00 FF
;   0x00F61A..0x00F627  86 85 06 05 84 83 82 04 03 02 81 80 01 FF
;   0x00F628..0x00F632  86 85 06 05 84 83 82 04 03 02 FF
; The last three are referenced by the 0x00F633 table and consumed by
; NoteChain_FindNode_A (0x02229A). A byte with bit7 set selects the GLOBAL pool-head
; array at 0x130F, indexed by (byte & 0x7F)*4; a byte with bit7 clear selects the
; part-descriptor-local array, indexed by byte*4.

[DATA] 0x00F633-0x00F692  Note-on layer descriptor table, 16 entries of 6 bytes
; Indexed by the low nibble of a note-event slot byte in NoteOn_Dispatch (0x0223BE).
; Entry layout: { u32 candidate_list_ptr; u8 priority_key; u8 node_param }.
;   entry 0  -> 0x0000F60B, key 0x00, param 0x03
;   entry 1  -> 0x0000F61A, key 0x01, param 0x03
;   entry 2  -> 0x0000F628, key 0x02, param 0x04
;   entries 3..15 -> 0x0000F628, key 0x02, param 0x05 (all identical)
; priority_key is passed to VoiceNode_PriorityList_Update; node_param is stored in
; (voice+0x26) at 0x022406.

[DATA] 0x00F693-0x00F6A6  PitchBend_Process jump table, 10 u16 offsets
; Offsets added to 0x022982 to form the target of the 'jp' at 0x02297D; the selector is
; WA-0x10 and must be in 0..9. In v142 nine entries are 0x0000 (they land on the
; 'ld HL,0 ; ret' stub at 0x022982) and only index 4 (selector 0x14) is 0x0004,
; reaching PitchBend_Process_Fallback at 0x022986.

[ROUTINE] 0x02219F AudioTick_UpdateVoice  confidence=high
; PER-TICK TONE-GENERATOR VOICE POLL. Services ONE bank of 16 hardware voices per
; call and is the ONLY place in the firmware where a voice is handed back to the
; software free pool.
; Caller: Audio_Process_Init (0x034CDB) at 0x034CED, on alternate ticks only - that
; routine toggles the byte at 0x041342 with 'xor 0xFF' and only calls us when it is
; non-zero. With the internal bank counter this means each of the 4 banks is polled
; once every 8 ticks.
; State: byte (0x1128) = bank counter, inc'd then masked to 0..3 on entry.
;   word (0x292E + bank*2) = shadow copy of the previous poll's active-voice bitmap.
;   word (0x2936 + bank*2) = software 'held / do not reap' bitmap, set by
;                            NoteOn_Dispatch (0x02243D) and cleared by
;                            VoiceSlot_Release (0x0225BB).
; Hardware: calls 0x021023 (ELF name DAC_Write_Sample - see [UNCERTAIN]) which writes
;   the TG register-address latch at 0x100000 and reads the word back. Latch value
;   0..3 returns the 16-bit active-voice bitmap for that bank; latch 0x180+ch returns
;   that channel's envelope level.
; Edge detection (0x2219F..0x02220A):
;     new  = TG_bitmap | held_bitmap        ; 0x221C5
;     hits = (new ^ prev) & prev            ; 0x221E9 / 0x221F1
;     prev = new                            ; 0x22200
;   so 'hits' are exactly the voices that were marked active last poll and are now
;   reported silent by the TG and are not software-held. Only those get reaped.
; Per-voice loop (see AudioTick_UpdateVoice_SlotLoop): voice index = bank*16 + i,
;   voice record = 0x148D + index*0x27 (39-byte records, 64 voices, 0x148D..0x1E4C).
; Exit: calls Voice_AdvanceSlotIterator (0x021BF5) with WA = bank.
; *** BUG-RELEVANT: if the tone generator never clears active bits (e.g. a sine-wave
; diagnostic renderer that has no end-of-sample / no auto-free), 'hits' is
; permanently zero, no voice is ever retired, and the free pool drains to empty. ***

[ROUTINE] 0x022340 NoteOn_Dispatch  confidence=high
; NOTE-ON: ALLOCATE UP TO FOUR VOICE NODES FOR ONE NOTE EVENT.
; In: XWA = note-event record (kept in local (XSP+0x10)):
;       +0x00 word  bits 0..6  = MIDI note number

;       +0x06 bit6  = 'continue chain' - if clear, the global chain head (0x1129) is
;                     reset to 0 first
;       +0x06..+0x09  per-slot bit7 = hold / retrigger flag

; layer limiter.
; Part descriptor = 0x1349 + part*0x0C. Each 0x00F633 entry is {u32 candidate-list ptr,
; u8 priority key, u8 node param}.
; *** BUG-RELEVANT: (voice+0x25) is forced to 0xFF here (0x0223FC) and the flag byte
; (voice+0x22) is set to 0x88 (held) or 0x08 (non-held) - bit2 is NOT set in either
; case. AudioTick_UpdateVoice's envelope-based release path requires level < 0x80 AND
; bit2, and its 'and A,0x81' guard also skips anything with bit7 set. So a voice that
